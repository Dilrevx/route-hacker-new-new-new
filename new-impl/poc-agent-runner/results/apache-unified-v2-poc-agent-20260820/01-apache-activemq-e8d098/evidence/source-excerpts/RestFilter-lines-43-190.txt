    43	/**
    44	 * <p>
    45	 * Adds support for HTTP PUT, MOVE and DELETE methods. If init parameters
    46	 * read-permission-role and write-permission-role are defined then all requests
    47	 * are authorized using the defined roles. Also GET methods are authorized.
    48	 * </p>
    49	 *
    50	 * @author Aleksi Kallio
    51	 */
    52	public class RestFilter implements Filter {
    53	    private static final Logger LOG = LoggerFactory.getLogger(RestFilter.class);
    54
    55	    private static final String HTTP_HEADER_DESTINATION = "Destination";
    56	    private static final String HTTP_METHOD_MOVE = "MOVE";
    57	    private static final String HTTP_METHOD_PUT = "PUT";
    58	    private static final String HTTP_METHOD_GET = "GET";
    59	    private static final String HTTP_METHOD_DELETE = "DELETE";
    60
    61	    private String readPermissionRole;
    62	    private String writePermissionRole;
    63	    private FilterConfig filterConfig;
    64
    65	    public void init(FilterConfig filterConfig) throws UnavailableException {
    66	        this.filterConfig = filterConfig;
    67	        readPermissionRole = filterConfig.getInitParameter("read-permission-role");
    68	        writePermissionRole = filterConfig.getInitParameter("write-permission-role");
    69	    }
    70
    71	    private File locateFile(HttpServletRequest request) {
    72	        return new File(filterConfig.getServletContext().getRealPath(request.getServletPath()), request.getPathInfo());
    73	    }
    74
    75	    public void doFilter(ServletRequest request, ServletResponse response, FilterChain chain) throws IOException, ServletException {
    76	        if (!(request instanceof HttpServletRequest && response instanceof HttpServletResponse)) {
    77	            if (LOG.isDebugEnabled()) {
    78	                LOG.debug("request not HTTP, can not understand: " + request.toString());
    79	            }
    80	            chain.doFilter(request, response);
    81	            return;
    82	        }
    83
    84	        HttpServletRequest httpRequest = (HttpServletRequest)request;
    85	        HttpServletResponse httpResponse = (HttpServletResponse)response;
    86
    87	        if (httpRequest.getMethod().equals(HTTP_METHOD_MOVE)) {
    88	            doMove(httpRequest, httpResponse);
    89	        } else if (httpRequest.getMethod().equals(HTTP_METHOD_PUT)) {
    90	            doPut(httpRequest, httpResponse);
    91	        } else if (httpRequest.getMethod().equals(HTTP_METHOD_GET)) {
    92	            if (checkGet(httpRequest, httpResponse)) {
    93	                chain.doFilter(httpRequest, httpResponse); // actual processing
    94	                                                            // done elsewhere
    95	            }
    96	        } else if (httpRequest.getMethod().equals(HTTP_METHOD_DELETE)) {
    97	            doDelete(httpRequest, httpResponse);
    98	        } else {
    99	            chain.doFilter(httpRequest, httpResponse);
   100	        }
   101	    }
   102
   103	    protected void doMove(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException {
   104	        if (LOG.isDebugEnabled()) {
   105	            LOG.debug("RESTful file access: MOVE request for " + request.getRequestURI());
   106	        }
   107
   108	        if (writePermissionRole != null && !request.isUserInRole(writePermissionRole)) {
   109	            response.sendError(HttpURLConnection.HTTP_FORBIDDEN);
   110	            return;
   111	        }
   112
   113	        File file = locateFile(request);
   114	        String destination = request.getHeader(HTTP_HEADER_DESTINATION);
   115
   116	        if (destination == null) {
   117	            response.sendError(HttpURLConnection.HTTP_BAD_REQUEST, "Destination header not found");
   118	            return;
   119	        }
   120
   121	        try {
   122	            URL destinationUrl = new URL(destination);
   123	            IOHelper.copyFile(file, new File(destinationUrl.getFile()));
   124	            IOHelper.deleteFile(file);
   125	        } catch (IOException e) {
   126	            response.sendError(HttpURLConnection.HTTP_INTERNAL_ERROR); // file
   127	                                                                        // could
   128	                                                                        // not
   129	                                                                        // be
   130	                                                                        // moved
   131	            return;
   132	        }
   133
   134	        response.setStatus(HttpURLConnection.HTTP_NO_CONTENT); // we return no
   135	                                                                // content
   136	    }
   137
   138	    protected boolean checkGet(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException {
   139	        if (LOG.isDebugEnabled()) {
   140	            LOG.debug("RESTful file access: GET request for " + request.getRequestURI());
   141	        }
   142
   143	        if (readPermissionRole != null && !request.isUserInRole(readPermissionRole)) {
   144	            response.sendError(HttpURLConnection.HTTP_FORBIDDEN);
   145	            return false;
   146	        } else {
   147	            return true;
   148	        }
   149	    }
   150
   151	    protected void doPut(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException {
   152	        if (LOG.isDebugEnabled()) {
   153	            LOG.debug("RESTful file access: PUT request for " + request.getRequestURI());
   154	        }
   155
   156	        if (writePermissionRole != null && !request.isUserInRole(writePermissionRole)) {
   157	            response.sendError(HttpURLConnection.HTTP_FORBIDDEN);
   158	            return;
   159	        }
   160
   161	        File file = locateFile(request);
   162
   163	        if (file.exists()) {
   164	            boolean success = file.delete(); // replace file if it exists
   165	            if (!success) {
   166	                response.sendError(HttpURLConnection.HTTP_INTERNAL_ERROR); // file
   167	                                                                            // existed
   168	                                                                            // and
   169	                                                                            // could
   170	                                                                            // not
   171	                                                                            // be
   172	                                                                            // deleted
   173	                return;
   174	            }
   175	        }
   176
   177	        FileOutputStream out = new FileOutputStream(file);
   178	        try {
   179	            IOHelper.copyInputStream(request.getInputStream(), out);
   180	        } catch (IOException e) {
   181	            LOG.warn("Exception occured" , e);
   182	            throw e;
   183	        } finally {
   184	            out.close();
   185	        }
   186
   187	        response.setStatus(HttpURLConnection.HTTP_NO_CONTENT); // we return no
   188	                                                                // content
   189	    }
